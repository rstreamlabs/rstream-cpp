cmake_minimum_required(VERSION 3.20)
project(ncurses_qualification LANGUAGES CXX)
find_package(Curses CONFIG REQUIRED)
add_executable(ncurses-qualification main.cpp)
target_compile_features(ncurses-qualification PRIVATE cxx_std_20)
target_link_libraries(ncurses-qualification PRIVATE Curses::Curses)
if(TEST_FULLY_STATIC)
  target_link_options(ncurses-qualification PRIVATE -static)
endif()
enable_testing()
add_test(NAME ncurses-runtime COMMAND ncurses-qualification)
set_tests_properties(ncurses-runtime PROPERTIES TIMEOUT 60)
